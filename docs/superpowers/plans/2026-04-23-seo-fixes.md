const FINANCE_SLUGS = ['mortgage', 'loan', 'salary', 'tax', 'roi', 'debt', 'savings', 'investment', 'inflation', 'compound', 'retirement']
const HEALTH_SLUGS = ['bmi', 'calorie', 'tdee', 'body-fat', 'weight', 'nutrition']
